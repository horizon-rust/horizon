// The light of colour `f` under its writer's mark `a` (hdr_mark) where the world `shows`, else
// the colour clipped at white; and whether a light source lifted it.
struct ReferenceLit {
    light: vec3<f32>,
    lifted: bool,
};

fn reference_lit(f: vec3<f32>, a: f32, shows: bool) -> ReferenceLit {
    var lifted = false;
    // Light over 1 shows only where the world does and carries no mark (1): light written
    // as light. Marked pixels clip at white before their lift, so light sources and the sun
    // keep the look they were tuned at; so do HDR Lighting off and the title.
    let c = select(min(f, vec3(1.0)), f, shows && a == 1.0);
    // Gamma 2.2, as SDR displays and the compositors' SDR paths decode it.
    var light = pow(c, vec3(2.2));
    // The mark is below 0.75 on light sources, the sun and the moon
    // (terrain.wgsl, entity.wgsl): its band of 0.175 is the tier, the lower the
    // brighter, and where in the band how much (hdr_mark).
    if shows && a < 0.75 {
        let band = u32(clamp(floor((a + 0.005) / 0.175), 0.0, 3.0));
        let tier = 3u - band;
        let lift = 1.0 - clamp((a - 0.175 * f32(band)) / 0.165, 0.0, 1.0);
        if lift > 0.0 {
            let gain = 1.0 + (params.peak - 1.0) * params.share[tier] * lift;
            light *= max(gain, 1.0);
            lifted = true;
        }
    }
    // HORIZON_HDR_DEBUG=1: what isn't lifted turns grey, what is keeps its
    // colour with a magenta tint, so a lift shows on any colour.
    if params.debug > 0.5 && params.world > 0.5 {
        if lifted {
            light *= vec3(1.0, 0.6, 1.0);
        } else {
            light = vec3(dot(light, vec3(0.2126, 0.7152, 0.0722)));
        }
    }
    return ReferenceLit(light, lifted);
}
